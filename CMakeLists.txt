cmake_minimum_required(VERSION 3.21)
project(dlssnr_pascal VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 and MSVC are required (Windows SDK and SEH).")
endif()
add_library(dlssnr_pascal STATIC src/runtime.cpp src/NeuralAdmission.cpp
 src/NeuralRuntimeKernels.cpp src/NeuralRuntimeKernelPlan.cpp src/NeuralRenderingRuntimeIdentity.cpp)
add_library(dlssnr_pascal::compat ALIAS dlssnr_pascal)
target_compile_features(dlssnr_pascal PUBLIC cxx_std_20)
target_compile_definitions(dlssnr_pascal PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(dlssnr_pascal PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_link_libraries(dlssnr_pascal PUBLIC dxgi bcrypt version)
option(DLSSNR_PASCAL_BUILD_TESTS "CPU-only tests; no GPU dispatch" ${PROJECT_IS_TOP_LEVEL})
set(DLSSNR_PASCAL_PTXAS "" CACHE FILEPATH "CUDA 12.9 V12.9.86 ptxas executable")
if(DLSSNR_PASCAL_BUILD_TESTS)
 enable_testing()
 find_package(Python3 REQUIRED COMPONENTS Interpreter)
 add_executable(admission_tests tests/NeuralAdmissionTests.cpp)
 target_link_libraries(admission_tests PRIVATE dlssnr_pascal)
 add_test(NAME admission COMMAND admission_tests)
 foreach(test IN ITEMS inventory launch_metadata launch_bounds sm_general)
  add_test(NAME ${test} COMMAND "${Python3_EXECUTABLE}" -I -B "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_${test}.py")
  set_tests_properties(${test} PROPERTIES ENVIRONMENT "TRP_TEST_PTXAS=${DLSSNR_PASCAL_PTXAS};TEMP=${CMAKE_CURRENT_BINARY_DIR};TMP=${CMAKE_CURRENT_BINARY_DIR}")
 endforeach()
 set_tests_properties(admission PROPERTIES ENVIRONMENT "TEMP=${CMAKE_CURRENT_BINARY_DIR};TMP=${CMAKE_CURRENT_BINARY_DIR}")
endif()
